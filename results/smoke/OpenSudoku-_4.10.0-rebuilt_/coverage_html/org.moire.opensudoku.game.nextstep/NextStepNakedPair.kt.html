package org.moire.opensudoku.game.nextstep

import android.content.Context
import org.moire.opensudoku.game.SudokuBoard

/** Strategy: Naked Pair / Locked Pair
 *
 *  The strategy "Naked Pair / Locked Pair" is implemented in the package "Naked Group".
 *  It is a "Naked Group" with size 2.
 */
class NextStepNakedPair(
	context: Context,
	board: SudokuBoard,
	hintLevel: HintLevels ): NextStepNakedGroup( context, board, hintLevel )
{

	override fun search(): Boolean {
		return checkForNakedGroup(2)
	}

}
