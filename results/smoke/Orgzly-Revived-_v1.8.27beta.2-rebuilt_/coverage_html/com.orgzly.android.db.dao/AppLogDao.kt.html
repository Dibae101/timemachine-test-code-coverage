import androidx.room.Dao

import kotlinx.coroutines.flow.Flow

@Dao
abstract class AppLogDao : BaseDao<AppLog> {
    @Query("SELECT * FROM app_logs ORDER BY timestamp")
    abstract fun getFlow(): Flow<List<AppLog>>
}
