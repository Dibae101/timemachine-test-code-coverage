package org.quantumbadger.redreader.common;

public enum NeverAlwaysOrWifiOnly {
	NEVER {
		@Override
		public boolean isEnabled(final boolean wifiActive) {
			return false;
		}
	},
	WIFIONLY {
		@Override
		public boolean isEnabled(final boolean wifiActive) {
			return wifiActive;
		}
	},
	ALWAYS {
		@Override
		public boolean isEnabled(final boolean wifiActive) {
			return true;
		}
	};

	public abstract boolean isEnabled(final boolean wifiActive);
}
