@Composable
fun Icon(
    icon: String,
    modifier: Modifier = Modifier,
    size: TextUnit? = 24.sp,
    color: Color? = null,
) {
    val iconInfo = IIconInfo.resolveIcon(icon)
    if (iconInfo == null) {
        Text(color = Color.Red, text = icon, modifier = modifier)
    } else {
        Icon(iconInfo, modifier, size, color)
    }
}

@Composable
fun Icon(
    iconInfo: IIconInfo,
    modifier: Modifier = Modifier,
    size: TextUnit? = 24.sp,
    color: Color? = null,
) {
    when (iconInfo) {
        is ExtraIcon -> {
            Icon(
                modifier = modifier.optional(size) {
                    size(it)
                },
                painter = rememberDrawablePainter(drawable = iconInfo.asDrawable(LocalContext.current)),
                contentDescription = stringResource(id = iconInfo.label),
                tint = color ?: LocalContentColor.current
            )
        }

        is IconInfo -> {
            CharIcon(
                char = iconInfo.unicode,
                modifier = modifier,
                fontFamily = iconInfo.fontFamily,
                size = size,
                color = color
            )
        }
    }
}

/**
 * if @param size is null, use AutoSize
 */
@Composable
fun CharIcon(
    char: Char,
    modifier: Modifier = Modifier,
    fontFamily: FontFamily? = null,
    size: TextUnit? = 24.sp,
    color: Color? = null,
) {
    if (size == null) {
        BasicText(
            modifier = modifier,
            text = char.toString(),
            style = TextStyle(
                fontFamily = fontFamily,
                color = color ?: LocalContentColor.current
            ),
            autoSize = TextAutoSize.StepBased(minFontSize = 6.sp)
        )
    } else {
        Text(
            modifier = modifier,
            text = char.toString(),
            style = TextStyle(
                lineHeight = size
            ),
            fontFamily = fontFamily,
            fontSize = size,
            color = color ?: Color.Unspecified
        )
    }
}
