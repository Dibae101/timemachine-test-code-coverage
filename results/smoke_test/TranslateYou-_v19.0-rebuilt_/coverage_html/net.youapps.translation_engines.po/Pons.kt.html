package net.youapps.translation_engines.po

import net.youapps.translation_engines.po.obj.PonsData
import net.youapps.translation_engines.po.obj.PonsLanguages
import retrofit2.http.Body
import retrofit2.http.GET
import retrofit2.http.POST
import retrofit2.http.Query

interface Pons {
    @GET("text-translation-web/v4/languages")
    suspend fun getLanguages(
        @Query("locale") locale: String = "en"
    ): PonsLanguages

    @POST("text-translation-web/v4/translate")
    suspend fun translate(
        @Query("locale") locale: String = "en",
        @Body body: PonsData
    ): PonsData
}
