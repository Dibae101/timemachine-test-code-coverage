package org.moire.opensudoku.game.command

class FillInMarksCommand : AbstractMultiMarksCommand() {
	override fun execute() {
		val cells = cells
		primaryMarksBeforeCommandEntries.clear()
		saveOldMarks()
		cells.fillInPrimaryMarks()
	}
}
