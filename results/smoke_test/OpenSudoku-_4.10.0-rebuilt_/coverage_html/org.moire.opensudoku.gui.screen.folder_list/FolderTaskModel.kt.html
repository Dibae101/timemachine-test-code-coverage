package org.moire.opensudoku.gui.screen.folder_list

import android.content.Context

abstract class FolderTaskModel {
	var maxValue = 0
	var currentValue = 0
	var corruptedCount = 0
	var titleParam = ""
	var isCancelled = false

	abstract suspend fun run(context: Context, onTaskFinished: (Long, Boolean, String) -> Unit)

	fun cancel() {
		isCancelled = true
	}
}
